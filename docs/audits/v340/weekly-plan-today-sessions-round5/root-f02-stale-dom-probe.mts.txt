import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { chromium, type Route } from 'playwright';

const cwd = process.cwd();
assert.ok(cwd.includes('mindx-weekly-plan-audit-r2-r3-'));
assert.ok(process.env.MINDX_DB_PATH?.startsWith(cwd));
assert.ok(process.env.MINDX_RUNNING_CLASSES_PATH?.startsWith(cwd));
const { getDb, closeDb } = await import('../src/lib/db/sqlite');
const { ClassSyncer } = await import('../src/lib/playwright/sync-class');
const { BrowserManager } = await import('../src/lib/playwright/browser-manager');
const { authService } = await import('../src/lib/playwright/auth');
assert.equal(getDb().name, process.env.MINDX_DB_PATH);

const browser = await chromium.launch({ headless: true });
const context = await browser.newContext({ serviceWorkers: 'block' });
const page = await context.newPage();
await page.addInitScript('window.__name=(fn)=>fn');
const bm = BrowserManager.getInstance();
const originalPage = bm.getPage, originalAuth = authService.ensureAuthenticated;
bm.getPage = async () => page;
const results: any[] = [];
const codes = ['VT-QA-JSB601', 'VT-QA-JSB602', 'VT-QA-JSB603'];
const row = (code: string) => `<tr><td>1</td><td>${code}</td><td>JSB</td><td>VT</td><td>Running</td></tr>`;

try {
  for (const mode of ['stale-filtered-response'] as const) {
    let elapsed = 0;
    let pending: Route | undefined;
    let markSeen!: () => void;
    const routeSeen = new Promise<void>(resolve => { markSeen = resolve; });
    const requests: any[] = [];
    const prior = codes.slice(0, mode === 'valid-same-table' ? 2 : 3).map(classCode => ({ classCode, status: 'Running' }));
    const tableHtml = codes.slice(0, 2).map(row).join('');
    const operationName = mode === 'non-class-response' ? 'getMe' : 'getClasses';
    const variables = mode === 'stale-filtered-response' ? { search: 'old' } : mode === 'valid-same-table' ? { search: '' } : {};
    const responseBody = mode === 'non-class-response'
      ? { data: { me: { id: 'SYNTHETIC_TEACHER' } } }
      : { data: { classes: codes.slice(0, 1).map(classCode => ({ classCode, status: 'Running' })) } };
    await context.unrouteAll({ behavior: 'wait' });
    await context.route('**/*', async route => {
      if (route.request().resourceType() === 'document') {
        await route.fulfill({ status: 200, contentType: 'text/html', body: '<meta charset="utf-8"><body></body>' });
      } else if (route.request().url() === 'https://lms-api.mindx.edu.vn/graphql') {
        requests.push({ url: route.request().url(), method: route.request().method(),
          postData: route.request().postDataJSON(), searchValueAtRequest: await page.locator('input').inputValue(),
          elapsedAtRequest: elapsed });
        pending = route;
        markSeen();
      } else {
        await route.abort();
      }
    });
    await page.goto('https://lms.mindx.edu.vn/admin/classes');
    await page.setContent(`<input placeholder="Search classes" value="old"><table><tbody>${tableHtml}</tbody></table>`);
    await page.evaluate(({ operationName, variables, mode, tableHtml }) => {
      (window as any).__done = false;
      (window as any).__start = () => {
        void fetch('https://lms-api.mindx.edu.vn/graphql', {
          method: 'POST', headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ operationName, variables }),
        }).then(r => r.json()).then(() => {
          if (mode === 'stale-filtered-response') document.querySelector('tbody tr:last-child')!.remove();
          (window as any).__done = true;
        });
      };
      if (mode !== 'stale-filtered-response') {
        document.querySelector('input')!.addEventListener('keydown', e => {
          if ((e as KeyboardEvent).key === 'Enter') (window as any).__start();
        });
      }
    }, { operationName, variables, mode, tableHtml });
    getDb().prepare('DELETE FROM classes_sessions').run();
    fs.writeFileSync(process.env.MINDX_RUNNING_CLASSES_PATH!, JSON.stringify(prior));
    const beforeRaw = fs.readFileSync(process.env.MINDX_RUNNING_CLASSES_PATH!, 'utf8');
    const beforeText = await page.locator('tbody').innerText();
    authService.ensureAuthenticated = async () => {
      if (mode === 'stale-filtered-response') {
        await page.evaluate(() => (window as any).__start());
        await routeSeen;
      }
      return true;
    };
    const responseListenerCountBefore = page.listeners('response').length;
    page.waitForTimeout = async (ms: number) => {
      elapsed += ms;
      if (elapsed === 400) {
        await routeSeen;
        assert.ok(pending);
        requests[0].searchValueAtResponse = await page.locator('input').inputValue();
        requests[0].elapsedAtResponse = elapsed;
        requests[0].responseStatus = 200;
        requests[0].responseBody = responseBody;
        await pending.fulfill({ status: 200, contentType: 'application/json',
          headers: { 'Access-Control-Allow-Origin': '*' }, body: JSON.stringify(responseBody) });
        await page.waitForFunction(() => (window as any).__done === true);
      }
    };
    const result = await ClassSyncer.fetchRunningClasses({ persistToConfig: true });
    const afterRaw = fs.readFileSync(process.env.MINDX_RUNNING_CLASSES_PATH!, 'utf8');
    const cache = JSON.parse(afterRaw);
    const observed = { elapsedMs: elapsed, simulatedWaits: true, result,
      cacheBeforeLength: prior.length, cacheAfterLength: cache.length, cacheUnchangedBytes: beforeRaw === afterRaw,
      inputAfter: await page.locator('input').inputValue(),
      tbodyTextUnchanged: beforeText === await page.locator('tbody').innerText(), requests,
      observerRemoved: page.listeners('response').length === responseListenerCountBefore };
    let status = 'PASS', error: string | undefined;
    try {
      assert.equal(observed.inputAfter, '');
      assert.equal(observed.tbodyTextUnchanged, false);
      assert.equal(observed.observerRemoved, true);
      if (mode === 'valid-same-table') {
        assert.equal(result.success, true, 'A valid post-clear class response can confirm an unchanged table');
        assert.equal(cache.length, 2);
      } else {
        assert.equal(cache.length, 3, 'A non-class or pre-clear filtered response must not authorize stale 2-row persistence');
        assert.equal(result.success, false, 'No valid clear-search refresh completed');
        assert.equal(beforeRaw, afterRaw);
      }
    } catch (e: any) { status = 'FAIL'; error = e.message; }
    const item = { id: `F02-R4-stale-response-changes-dom`, status, ...(error ? { error } : {}), observed };
    results.push(item);
    console.log(JSON.stringify(item));
  }
} finally {
  bm.getPage = originalPage;
  authService.ensureAuthenticated = originalAuth;
  await browser.close();
  closeDb();
}
const summary = { pass: results.filter(x => x.status === 'PASS').length,
  fail: results.filter(x => x.status === 'FAIL').length, total: results.length, results };
fs.writeFileSync(path.join(cwd, 'audit-evidence/root-f02-stale-dom-results.json'), JSON.stringify(summary, null, 2));
console.log('ROOT F02 STALE DOM SUMMARY', JSON.stringify({ pass: summary.pass, fail: summary.fail, total: summary.total }));
process.exitCode = summary.fail ? 1 : 0;
