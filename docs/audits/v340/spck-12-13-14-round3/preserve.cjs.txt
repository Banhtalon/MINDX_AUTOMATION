const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const root=__dirname,state=JSON.parse(fs.readFileSync(path.join(root,'state.json'),'utf8')),repo=state.repo,work=state.work;
const dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round3');
if(fs.existsSync(dest))throw Error('Evidence destination already exists');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const before=JSON.parse(fs.readFileSync(path.join(root,'source-manifest-before.json'),'utf8'));
const after=Object.fromEntries(Object.keys(before).map(p=>[p,sha(path.join(repo,p))]));
const sourceChanged=Object.keys(before).filter(p=>before[p]!==after[p]);
const snapshotChanged=Object.keys(before).filter(p=>before[p]!==sha(path.join(work,p)));
const patch=cp.execFileSync('git',['diff','--binary','HEAD'],{cwd:repo,maxBuffer:20*1024*1024});
if(!patch.equals(fs.readFileSync(path.join(root,'working-tree.patch'))))throw Error('Original tracked diff changed');
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(sourceChanged.length||snapshotChanged.length||Object.keys(state.expected).some(p=>state.expected[p]!==productionAfter[p]))throw Error('Preservation mismatch');
fs.writeFileSync(path.join(root,'source-manifest-after.json'),JSON.stringify(after,null,2));
const gates=['tsc','v270','v272','s5','probe','supplementary','callback','focused-final'].map(g=>JSON.parse(fs.readFileSync(path.join(root,g+'-result.json'),'utf8')));
const read=p=>JSON.parse(fs.readFileSync(path.join(root,p),'utf8'));
const probe=read('probe-results.json'),supp=read('supplementary-results.json'),callback=read('callback-results.json'),focused=read('focused-results.json');
const closed=['F03','F04','F05','F06','F09','F10'],statuses=Object.fromEntries(Array.from({length:10},(_,i)=>'F'+String(i+1).padStart(2,'0')).map(id=>[id,closed.includes(id)?'CLOSED':'OPEN']));
fs.writeFileSync(path.join(root,'closure-results.json'),JSON.stringify({verdict:'NO-GO',closed:6,open:4,statuses,blockingProbes:focused.results.filter(r=>r.status==='FAIL').map(r=>({id:r.id,finding:r.finding,label:r.label}))},null,2));
const observations=focused.results.map(r=>({id:r.id,finding:r.finding,status:r.status,label:r.label,source:r.observed?.source,tags:r.observed?.tags,flags:r.observed?.absentWebFeatures,summary:r.observed?.summaryText,comment:r.observed?.formattedTeacherComment,commentStatus:r.observed?.status}));
fs.writeFileSync(path.join(root,'focused-observations.json'),JSON.stringify(observations,null,2));
const networkFiles=fs.readdirSync(root).filter(n=>n.endsWith('-network.jsonl'));
const events=networkFiles.flatMap(n=>fs.readFileSync(path.join(root,n),'utf8').trim().split('\n').filter(Boolean).map(line=>({log:n,...JSON.parse(line)})));
const versions=Object.fromEntries(['typescript','tsx','next','better-sqlite3','esbuild','playwright'].map(p=>[p,JSON.parse(fs.readFileSync(path.join(repo,'node_modules',p,'package.json'),'utf8')).version]));
const summary={...state,verdict:'NO-GO',mandatoryGateCount:6,mandatoryCommandCount:7,mandatoryPassed:gates.slice(0,7).every(g=>g.exitCode===0),gates,tests:{round1:{total:probe.total,passed:probe.passed,failed:probe.failed},round2:{total:supp.total,passed:supp.passed,failed:supp.failed,errors:supp.errors},F09:{total:callback.total,passed:callback.passed,failed:callback.failed},round3:{total:focused.total,passed:focused.passed,failed:focused.failed,errors:focused.errors}},statuses,versions,productionAfter,productionMatchesOwner:true,sourceChanged,snapshotChanged,trackedDiffUnchanged:true,network:{logs:networkFiles,events:events.length,blocked:events.filter(e=>e.kind.startsWith('blocked')),sqliteOpened:events.filter(e=>e.kind==='sqlite-open').map(e=>e.host),limitation:'Instrumentation only; not an operating-system packet capture'},initialFocusedAttempt:'Two fixture setup errors due to a missing synthetic session 13; corrected and rerun on a fresh DB. Initial attempt retained, superseded by focused-final; no product code changed.',scope:'Only requested audit report and Round 3 evidence are written in original workspace; synthetic and mocked services only.'};
fs.writeFileSync(path.join(root,'verification-summary.json'),JSON.stringify(summary,null,2));
for(const name of ['independent-probe.ts','supplementary-probe.ts','round3-focused-probe.ts'])fs.copyFileSync(path.join(work,'scratch',name),path.join(root,name+'.txt'));
fs.mkdirSync(dest,{recursive:false});
for(const name of fs.readdirSync(root).filter(n=>fs.statSync(path.join(root,n)).isFile())){
  const output=/\.(?:cjs|log)$/.test(name)?name+'.txt':name;
  fs.copyFileSync(path.join(root,name),path.join(dest,output));
  if(sha(path.join(root,name))!==sha(path.join(dest,output)))throw Error('Evidence copy mismatch '+name);
}
console.log(JSON.stringify({dest,mandatoryPassed:summary.mandatoryPassed,tests:summary.tests,statuses,sourceCount:state.sourceCount,sourceChanged,snapshotChanged,blockedEvents:summary.network.blocked},null,2));
