const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const root=__dirname,state=JSON.parse(fs.readFileSync(path.join(root,'state.json'),'utf8')),repo=state.repo,work=state.work;
const dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round5');if(fs.existsSync(dest))throw Error('Evidence destination exists');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const before=JSON.parse(fs.readFileSync(path.join(root,'source-manifest-before.json'),'utf8'));
const after=Object.fromEntries(Object.keys(before).map(p=>[p,sha(path.join(repo,p))]));
const sourceChanged=Object.keys(before).filter(p=>before[p]!==after[p]);
const snapshotChanged=Object.keys(before).filter(p=>before[p]!==sha(path.join(work,p)));
const patch=cp.execFileSync('git',['diff','--binary','HEAD'],{cwd:repo,maxBuffer:20*1024*1024,stdio:['ignore','pipe','pipe']});
if(!patch.equals(fs.readFileSync(path.join(root,'working-tree.patch'))))throw Error('Original tracked diff changed');
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(sourceChanged.length||snapshotChanged.length||Object.keys(state.expected).some(p=>state.expected[p]!==productionAfter[p]))throw Error('Preservation mismatch');
fs.writeFileSync(path.join(root,'source-manifest-after.json'),JSON.stringify(after,null,2));
const gates=['tsc','v270','v272','s5','probe','supplementary','callback','focused','closure','review'].map(g=>JSON.parse(fs.readFileSync(path.join(root,g+'-result.json'),'utf8')));
const read=p=>JSON.parse(fs.readFileSync(path.join(root,p),'utf8'));
const tests=Object.fromEntries([['round1','probe-results.json'],['round2','supplementary-results.json'],['F09','callback-results.json'],['round3','focused-results.json'],['round4','round4-closure-results.json'],['round5','round5-focused-results.json']].map(([k,p])=>{const r=read(p);return [k,{total:r.total,passed:r.passed,failed:r.failed,errors:r.errors??0}]}));
const closed=['F02','F03','F04','F05','F06','F09','F10'];
const statuses=Object.fromEntries(Array.from({length:10},(_,i)=>'F'+String(i+1).padStart(2,'0')).map(id=>[id,closed.includes(id)?'CLOSED':'OPEN']));
const results=read('round5-focused-results.json').results;
const mandatoryPassed=gates.slice(0,9).every(g=>g.exitCode===0);
const verdict=mandatoryPassed&&results.every(r=>r.status==='PASS')?'GO':'NO-GO';
fs.writeFileSync(path.join(root,'closure-results.json'),JSON.stringify({verdict,closed:closed.length,open:10-closed.length,statuses,mandatoryPassed,blockingProbes:results.filter(r=>r.status==='FAIL').map(r=>({id:r.id,finding:r.finding,label:r.label})),note:'All 76 original Round 1-4 probe cases pass. Five supplementary Round 5 failures are variants of F01/F07/F08; not five new findings. No product changes.'},null,2));
const observations=results.map(r=>({id:r.id,finding:r.finding,status:r.status,label:r.label,source:r.observed?.source,tags:r.observed?.tags,flags:r.observed?.absentWebFeatures,summary:r.observed?.summaryText,dbReadBack:r.observed?.dbReadBack,displayedText:r.observed?.displayedText,runtimeCalls:r.observed?.runtimeCalls,htmlEntry:r.observed?.htmlEntry,readEntry:r.observed?.readEntry,normalizedReferencedPath:r.observed?.normalizedReferencedPath}));
fs.writeFileSync(path.join(root,'round5-focused-observations.json'),JSON.stringify(observations,null,2));
const networkFiles=fs.readdirSync(root).filter(n=>n.endsWith('-network.jsonl'));
const events=networkFiles.flatMap(n=>fs.readFileSync(path.join(root,n),'utf8').trim().split('\n').filter(Boolean).map(line=>({log:n,...JSON.parse(line)})));
const versions=Object.fromEntries(['typescript','tsx','next','better-sqlite3','esbuild','playwright'].map(p=>[p,JSON.parse(fs.readFileSync(path.join(repo,'node_modules',p,'package.json'),'utf8')).version]));
const reportPath=path.join(repo,'docs/audits/v340/CODEX_SPCK_12_13_14_AUDIT.md');
const originalHistory=fs.readFileSync(path.join(root,'round4-report.md.txt'));
if(!fs.readFileSync(reportPath).equals(originalHistory))throw Error('Report already changed unexpectedly');
const probeHashesAfter=Object.fromEntries(Object.entries(state.probeHashes).map(([name,hashes])=>{const p=name.endsWith('.ts')?path.join(work,'scratch',name):path.join(root,name);const actual=sha(p);if(actual!==hashes.source||hashes.source!==hashes.executed)throw Error('Probe changed '+name);return [name,actual]}));
const summary={...state,verdict,mandatoryGateCount:8,mandatoryCommandCount:9,mandatoryPassed,gates,tests,statuses,versions,productionAfter,productionMatchesOwner:true,sourceChanged,snapshotChanged,trackedDiffUnchanged:true,probeHashesAfter,history:{originalBytes:originalHistory.length,originalSha256:sha(path.join(root,'round4-report.md.txt'))},network:{logs:networkFiles,events:events.length,blocked:events.filter(e=>e.kind.startsWith('blocked')),sqliteOpened:events.filter(e=>e.kind==='sqlite-open').map(e=>e.host),limitation:'In-process instrumentation only; not an operating-system packet capture'},reviewer:{agent:'/root/closure_source_review',scope:'Independent source inspection only; runtime results executed by primary auditor',requestedModel:'claude-sonnet-5-5-high',requestedModelAvailable:false,actual:'Current collaboration agent; no claim to have run the requested unavailable model',verdict:'NO-GO'},scope:'Only requested audit report and Round 5 evidence written in original workspace; synthetic/mocked services only. No source edits, commit, push or real-service acceptance.'};
fs.writeFileSync(path.join(root,'verification-summary.json'),JSON.stringify(summary,null,2));
for(const name of ['independent-probe.ts','supplementary-probe.ts','round3-focused-probe.ts','round4-closure-probe.ts','round5-focused-probe.ts'])fs.copyFileSync(path.join(work,'scratch',name),path.join(root,name+'.txt'));
fs.writeFileSync(path.join(root,'review-source-anchors.txt'),cp.execFileSync('rg',['-n','analyzeJsCodeSyntax|readFsJsPaths|readZipMountedJsPaths|readZipRelJsPaths|zipEntryDir|activeForbiddenPatterns|inspectZipArchiveSafely|styleMatch|initializeApp|prefixWithExt|mergedLinksMap|normStuName','src/app/api/spck/scan/route.ts','src/lib/gemini/spck-comment.ts','src/lib/playwright/spck-uploader.ts'],{cwd:work,encoding:'utf8'}));
fs.mkdirSync(dest);
const outputName=n=>/\.(?:cjs|log)$/.test(n)?n+'.txt':n;
for(const name of fs.readdirSync(root).filter(n=>fs.statSync(path.join(root,n)).isFile())){fs.copyFileSync(path.join(root,name),path.join(dest,outputName(name)));if(sha(path.join(root,name))!==sha(path.join(dest,outputName(name))))throw Error('Evidence copy mismatch '+name);}
console.log(JSON.stringify({dest,verdict,mandatoryPassed,tests,statuses,sourceCount:state.sourceCount,sourceChanged,snapshotChanged,blockedEvents:summary.network.blocked,versions,history:summary.history},null,2));
