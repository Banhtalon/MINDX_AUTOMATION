const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const root=__dirname,repo=process.argv[2],r8=path.join(repo,'docs/audits/v340/spck-12-13-14-round8');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
let s=fs.readFileSync(path.join(r8,'setup.cjs.txt'),'utf8');
s=s.replaceAll('spck-12-13-14-round8','spck-12-13-14-round9').replaceAll('Round 8','Round 9');
s=s.replace("r7=path.join(base,'spck-12-13-14-round7')", "r7=path.join(base,'spck-12-13-14-round8')");
s=s.replace("['round7-focused-probe.ts',path.join(r7,'round7-focused-probe.ts.txt')]]", "['round7-focused-probe.ts',path.join(r7,'round7-focused-probe.ts.txt')],['round8-context-probe.ts',path.join(r7,'round8-context-probe.ts.txt')]]");
s=s.replaceAll('changedSinceRound7','changedSinceRound8').replaceAll('source-changes-since-round7','source-changes-since-round8').replaceAll('round7-report.md.txt','round8-report.md.txt');
fs.writeFileSync(path.join(root,'setup.cjs'),s);
require('./setup.cjs');
let run=fs.readFileSync(path.join(r8,'run-all.cjs.txt'),'utf8').replace("['round7'", "['context','round7'");
fs.writeFileSync(path.join(root,'run-all.cjs'),run);
const walk=(d,r='')=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name),r+e.name+'/'):[r+e.name]);
const prior=Object.fromEntries([1,2,3,4,5,6,7,8].flatMap(n=>{const rel='docs/audits/v340/spck-12-13-14-round'+n;return walk(path.join(repo,rel)).map(p=>[rel+'/'+p,sha(path.join(repo,rel,p))])}));
fs.writeFileSync(path.join(root,'prior-evidence-manifest.json'),JSON.stringify(prior,null,2));
if(!fs.existsSync(path.join(root,'work/scratch/round8-context-probe.ts')))throw Error('Round 8 probe missing');
console.log('ROUND9_READY '+root);
