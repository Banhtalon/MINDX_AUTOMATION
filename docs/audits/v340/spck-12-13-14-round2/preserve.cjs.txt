const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const root=__dirname,state=JSON.parse(fs.readFileSync(path.join(root,'state.json'),'utf8')),repo=state.repo,work=state.work;
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const before=JSON.parse(fs.readFileSync(path.join(root,'source-manifest-before.json'),'utf8'));
const after=Object.fromEntries(Object.keys(before).map(p=>[p,sha(path.join(repo,p))]));
const changed=Object.keys(before).filter(p=>before[p]!==after[p]);const copyDiff=Object.keys(before).filter(p=>before[p]!==sha(path.join(work,p)));
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
const productionMatches=Object.keys(state.expected).every(p=>productionAfter[p]===state.productionBefore[p]&&productionAfter[p]===state.expected[p]);
if(changed.length||copyDiff.length||!productionMatches)throw Error(JSON.stringify({changed,copyDiff,productionMatches}));
const gates=['tsc','v270','v272','s5','probe'].map(g=>JSON.parse(fs.readFileSync(path.join(root,g+'-result.json'),'utf8')));
const probe=JSON.parse(fs.readFileSync(path.join(root,'probe-results.json'),'utf8'));
const supplementary=JSON.parse(fs.readFileSync(path.join(root,'supplementary-results.json'),'utf8'));
const callback=JSON.parse(fs.readFileSync(path.join(root,'callback-results.json'),'utf8'));
const networkLogs=fs.readdirSync(root).filter(p=>p.endsWith('-network.jsonl'));
const networkEvents=networkLogs.flatMap(p=>fs.readFileSync(path.join(root,p),'utf8').trim().split(/\r?\n/).filter(Boolean).map(l=>({log:p,...JSON.parse(l)})));
const blockedEvents=networkEvents.filter(e=>e.kind.startsWith('blocked'));
const mapping={F01:{status:'OPEN',priority:'P1',originalCases:['TAGS-02','TAGS-03'],extraCases:['R2-F01-01','R2-F01-02','R2-F01-03']},F02:{status:'OPEN',priority:'P1',originalCases:['HALLUCINATION-01','HALLUCINATION-02','HALLUCINATION-03'],extraCases:['R2-F02-01','R2-F02-02','R2-F02-03']},F03:{status:'CLOSED',priority:'P1',originalCases:['ZIP-04']},F04:{status:'CLOSED',priority:'P2',originalCases:['DRIVE-01','DRIVE-02','DRIVE-03']},F05:{status:'CLOSED',priority:'P2',originalCases:['DRIVE-04','DRIVE-05']},F06:{status:'CLOSED',priority:'P2',originalCases:['ROB-01','ROB-02']},F07:{status:'OPEN',priority:'P2',originalCases:['WEB-03','WEB-05'],extraCases:['R2-F07-01','R2-F07-02']},F08:{status:'OPEN',priority:'P2',originalCases:['WEB-04'],extraCases:['R2-F08-01','R2-F08-02']},F09:{status:'CLOSED',priority:'P2',originalCases:['LMS-01','LMS-02','LMS-03'],extraCases:['R2-F09-01','R2-F09-02']},F10:{status:'CLOSED',priority:'P2',originalCases:['SYNC-05','SYNC-06']}};
const probeSourceMatch=sha(path.join(work,'scratch/independent-probe.ts'))===sha(path.join(repo,'docs/audits/v340/spck-12-13-14-round1/independent-probe.ts.txt'));
const summary={...state,verifiedAt:new Date().toISOString(),gates,originalProbe:{total:probe.total,passed:probe.passed,failed:probe.failed,sourceByteIdenticalToRound1:probeSourceMatch},supplementary:{total:supplementary.total,passed:supplementary.passed,failed:supplementary.failed,errors:supplementary.errors},callback:{total:callback.total,passed:callback.passed,failed:callback.failed},sourcePreservation:{count:Object.keys(before).length,applicationAndScriptsCount:Object.keys(before).filter(p=>/^(src|scripts)\//.test(p)).length,changed,copyDiff},productionAfter,productionMatches,blockedEvents,findings:mapping,verdict:'NO-GO'};
fs.writeFileSync(path.join(root,'source-manifest-after.json'),JSON.stringify(after,null,2));
fs.writeFileSync(path.join(root,'closure-results.json'),JSON.stringify(mapping,null,2));
fs.writeFileSync(path.join(root,'verification-summary.json'),JSON.stringify(summary,null,2));
fs.writeFileSync(path.join(root,'source-changes-since-round1.json'),JSON.stringify(state.changedSinceRound1.map(p=>({file:p,sha256:before[p]})),null,2));
fs.writeFileSync(path.join(root,'working-tree.patch'),cp.execFileSync('git',['diff','--binary','HEAD','--','src','scripts'],{cwd:repo}));
fs.writeFileSync(path.join(root,'supplementary-observations.json'),JSON.stringify(supplementary.results.map(r=>({...r,observed:r.observed?.formattedTeacherComment!==undefined?{status:r.observed.status,formattedTeacherComment:r.observed.formattedTeacherComment}:r.observed})),null,2));
const dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round2');if(fs.existsSync(dest))throw Error('Evidence destination already exists');fs.mkdirSync(dest,{recursive:true});
for(const p of fs.readdirSync(root)){if(p==='work')continue;if(fs.statSync(path.join(root,p)).isFile()){const name=/\.(?:cjs|md)$/.test(p)?p+'.txt':p;fs.copyFileSync(path.join(root,p),path.join(dest,name));}}
for(const p of ['independent-probe.ts','supplementary-probe.ts'])fs.copyFileSync(path.join(work,'scratch',p),path.join(dest,p+'.txt'));
console.log(JSON.stringify({dest,sourcePreservation:summary.sourcePreservation,productionMatches,blockedEvents,probeSourceMatch,verdict:summary.verdict,files:fs.readdirSync(dest).length},null,2));
