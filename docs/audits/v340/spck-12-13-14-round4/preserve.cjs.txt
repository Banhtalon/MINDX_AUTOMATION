const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const root=__dirname,state=JSON.parse(fs.readFileSync(path.join(root,'state.json'),'utf8')),repo=state.repo,work=state.work;
const dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round4');if(fs.existsSync(dest))throw Error('Evidence destination exists');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const before=JSON.parse(fs.readFileSync(path.join(root,'source-manifest-before.json'),'utf8'));
const after=Object.fromEntries(Object.keys(before).map(p=>[p,sha(path.join(repo,p))]));
const sourceChanged=Object.keys(before).filter(p=>before[p]!==after[p]);
const snapshotChanged=Object.keys(before).filter(p=>before[p]!==sha(path.join(work,p)));
const patch=cp.execFileSync('git',['diff','--binary','HEAD'],{cwd:repo,maxBuffer:20*1024*1024});if(!patch.equals(fs.readFileSync(path.join(root,'working-tree.patch'))))throw Error('Original tracked diff changed');
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(sourceChanged.length||snapshotChanged.length||Object.keys(state.expected).some(p=>state.expected[p]!==productionAfter[p]))throw Error('Preservation mismatch');
fs.writeFileSync(path.join(root,'source-manifest-after.json'),JSON.stringify(after,null,2));
const gates=['tsc','v270','v272','s5','probe','supplementary','callback','focused','closure-final'].map(g=>JSON.parse(fs.readFileSync(path.join(root,g+'-result.json'),'utf8')));
const read=p=>JSON.parse(fs.readFileSync(path.join(root,p),'utf8'));
const probe=read('probe-results.json'),supp=read('supplementary-results.json'),callback=read('callback-results.json'),focused=read('focused-results.json'),closure=read('round4-closure-results.json');
const closed=['F03','F04','F05','F06','F09','F10'];const statuses=Object.fromEntries(Array.from({length:10},(_,i)=>'F'+String(i+1).padStart(2,'0')).map(id=>[id,closed.includes(id)?'CLOSED':'OPEN']));
fs.writeFileSync(path.join(root,'closure-results.json'),JSON.stringify({verdict:'NO-GO',closed:6,open:4,statuses,mandatoryPassed:gates.slice(0,8).every(g=>g.exitCode===0),blockingProbes:closure.results.filter(r=>r.status==='FAIL').map(r=>({id:r.id,finding:r.finding,label:r.label})),note:'All earlier Round 3 reproduction cases pass. Six additional cases across four existing findings still fail; no product changes were made.'},null,2));
const observations=closure.results.map(r=>({id:r.id,finding:r.finding,status:r.status,label:r.label,source:r.observed?.source,tags:r.observed?.tags,flags:r.observed?.absentWebFeatures,summary:r.observed?.summaryText,comment:r.observed?.formattedTeacherComment,commentStatus:r.observed?.status,persistedDraft:r.observed?.persistedDraft,runtimeCalls:r.observed?.runtimeCalls,unreadPath:r.observed?.unreadPath,unrelatedReadPath:r.observed?.unrelatedReadPath}));
fs.writeFileSync(path.join(root,'round4-closure-observations.json'),JSON.stringify(observations,null,2));
const networkFiles=fs.readdirSync(root).filter(n=>n.endsWith('-network.jsonl'));
const events=networkFiles.flatMap(n=>fs.readFileSync(path.join(root,n),'utf8').trim().split('\n').filter(Boolean).map(line=>({log:n,...JSON.parse(line)})));
const versions=Object.fromEntries(['typescript','tsx','next','better-sqlite3','esbuild','playwright'].map(p=>[p,JSON.parse(fs.readFileSync(path.join(repo,'node_modules',p,'package.json'),'utf8')).version]));
const count=r=>({total:r.total,passed:r.passed,failed:r.failed,errors:r.errors??0});
const summary={...state,verdict:'NO-GO',mandatoryGateCount:7,mandatoryCommandCount:8,mandatoryPassed:gates.slice(0,8).every(g=>g.exitCode===0),gates,tests:{round1:count(probe),round2:count(supp),F09:count(callback),round3:count(focused),round4:count(closure)},statuses,versions,productionAfter,productionMatchesOwner:true,sourceChanged,snapshotChanged,trackedDiffUnchanged:true,network:{logs:networkFiles,events:events.length,blocked:events.filter(e=>e.kind.startsWith('blocked')),sqliteOpened:events.filter(e=>e.kind==='sqlite-open').map(e=>e.host),limitation:'Instrumentation only; not an operating-system packet capture'},closureRuns:'Initial 15 cases produced 9 PASS and 6 FAIL. Final run adds template-expression runtime-call proof and read-back of saved synthetic drafts, using a fresh DB. Same 9 PASS / 6 FAIL / 0 ERROR; initial output retained.',scope:'Only requested audit report and Round 4 evidence are written in original workspace; synthetic and mocked services only. No source edits, commit, push or real-service acceptance.'};
fs.writeFileSync(path.join(root,'verification-summary.json'),JSON.stringify(summary,null,2));
for(const name of ['independent-probe.ts','supplementary-probe.ts','round3-focused-probe.ts','round4-closure-probe.ts'])fs.copyFileSync(path.join(work,'scratch',name),path.join(root,name+'.txt'));
fs.writeFileSync(path.join(root,'review-source-anchors.txt'),cp.execFileSync('rg',['-n','stripJsCommentsOnly|stripJsStringLiterals|readWebBasenames|cleanSrcBase|activeForbiddenPatterns|initializeApp|getDocs|display\\s','src/app/api/spck/scan/route.ts','src/lib/gemini/spck-comment.ts','src/lib/spck/spck-quick-tags.ts'],{cwd:work,encoding:'utf8'}));
fs.mkdirSync(dest,{recursive:false});
for(const name of fs.readdirSync(root).filter(n=>fs.statSync(path.join(root,n)).isFile())){const output=/\.(?:cjs|log)$/.test(name)?name+'.txt':name;fs.copyFileSync(path.join(root,name),path.join(dest,output));if(sha(path.join(root,name))!==sha(path.join(dest,output)))throw Error('Evidence copy mismatch '+name);}
console.log(JSON.stringify({dest,mandatoryPassed:summary.mandatoryPassed,tests:summary.tests,statuses,sourceCount:state.sourceCount,sourceChanged,snapshotChanged,blockedEvents:summary.network.blocked,versions},null,2));
