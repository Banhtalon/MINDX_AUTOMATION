const fs=require('node:fs'),path=require('node:path'),os=require('node:os'),crypto=require('node:crypto'),cp=require('node:child_process');
const repo=process.argv[2],dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round7');
const state=JSON.parse(fs.readFileSync(path.join(dest,'state.json'))),summary=JSON.parse(fs.readFileSync(path.join(dest,'verification-summary.json')));
const root=state.root,expectedRoot=path.join(os.tmpdir(),'codex-spck-121314-r7-b11d4193380f');
if(path.resolve(root)!==path.resolve(expectedRoot)||path.dirname(fs.realpathSync(root))!==fs.realpathSync(os.tmpdir())||path.resolve(process.cwd()).startsWith(path.resolve(root)))throw Error('Unsafe cleanup target or cwd');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const read=n=>JSON.parse(fs.readFileSync(path.join(dest,n)));
const before=read('source-manifest-before.json');
const walk=(dir,rel='')=>fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(dir,e.name),rel+e.name+'/'):[rel+e.name]);
const currentPaths=[...['src','scripts','public','data/config'].filter(d=>fs.existsSync(path.join(repo,d))).flatMap(d=>walk(path.join(repo,d)).map(p=>d+'/'+p)),...fs.readdirSync(repo,{withFileTypes:true}).filter(e=>e.isFile()&&!e.name.startsWith('.env')&&/\.(?:json|[cm]?js|[cm]?ts|tsx|css|md)$/.test(e.name)).map(e=>e.name)].sort();
if(JSON.stringify(currentPaths)!==JSON.stringify(Object.keys(before).sort()))throw Error('Source inventory changed');
const sourceChanged=Object.keys(before).filter(p=>sha(path.join(repo,p))!==before[p]);
if(sourceChanged.length||summary.snapshotChanged.length||summary.sourceChanged.length)throw Error('Source preservation failed');
const trackedDiff=cp.execFileSync('git',['diff','--binary','HEAD'],{cwd:repo,maxBuffer:20*1024*1024,stdio:['ignore','pipe','pipe']});
if(!trackedDiff.equals(fs.readFileSync(path.join(dest,'working-tree.patch.txt'))))throw Error('Tracked diff changed');
const prior=read('prior-evidence-manifest.json');
const priorEvidenceChanged=Object.keys(prior).filter(p=>sha(path.join(repo,p))!==prior[p]);
const priorPaths=[1,2,3,4,5,6].flatMap(r=>{const rel='docs/audits/v340/spck-12-13-14-round'+r;return walk(path.join(repo,rel)).map(p=>rel+'/'+p)}).sort();
if(priorEvidenceChanged.length||JSON.stringify(priorPaths)!==JSON.stringify(Object.keys(prior).sort()))throw Error('Prior evidence changed');
const reportPath=path.join(repo,'docs/audits/v340/CODEX_SPCK_12_13_14_AUDIT.md');
const old=fs.readFileSync(path.join(dest,'round6-report.md.txt'));
let report=fs.readFileSync(reportPath);
if(!report.subarray(report.length-old.length).equals(old))throw Error('Historical report changed');
let section=report.subarray(0,report.length-old.length).toString('utf8');
const cleanupNote='Thử dọn lần đầu gặp EPERM vì lệnh đang đứng trong thư mục snapshot trên Windows; junction đã được tháo trước đó. Đã chuyển vị trí chạy ra ngoài TEMP và dọn lại thành công; [mã hoàn tất dọn và xác minh](spck-12-13-14-round7/finish-cleanup.cjs.txt) ghi đầy đủ bước kiểm tra lại.\n\n';
const anchor='### R7.6. Kết luận và bàn giao';
if(!section.includes(anchor))throw Error('Missing report anchor');
section=section.replace(anchor,cleanupNote+anchor);
fs.writeFileSync(reportPath,Buffer.concat([Buffer.from(section,'utf8'),old]));
if(!section.includes('## VERDICT ROUND 7: NO-GO')||!section.includes('8/10 CLOSED; 2/10 OPEN'))throw Error('Wrong report verdict');
if(summary.gates.length!==12||summary.gates.slice(0,11).some(g=>g.exitCode!==0)||summary.gates[11].exitCode!==1||summary.network.blocked.length)throw Error('Wrong gates');
for(const [k,n] of Object.entries({round1:42,round2:10,F09:2,round3:7,round4:15,round5:8,round6:7})){const t=summary.tests[k];if(t.total!==n||t.passed!==n||t.failed||t.errors)throw Error('Probe mismatch '+k);}
if(summary.tests.round7.total!==4||summary.tests.round7.passed!==2||summary.tests.round7.failed!==2||summary.tests.round7.errors!==0)throw Error('Wrong R7 probes');
if(summary.verdict!=='NO-GO'||Object.values(summary.statuses).filter(s=>s==='CLOSED').length!==8||summary.statuses.F01!=='OPEN'||summary.statuses.F08!=='OPEN')throw Error('Wrong finding status');
for(const [name,pair] of Object.entries(state.probeHashes)){if(sha(path.join(dest,name+'.txt'))!==pair.source||pair.source!==pair.executed||summary.probeHashesAfter[name]!==pair.source)throw Error('Original probe changed '+name);}
const ignored=l=>l.includes('docs/audits/v340/CODEX_SPCK_12_13_14_AUDIT.md')||l.includes('docs/audits/v340/spck-12-13-14-round7/');
const filter=s=>s.split(/\r?\n/).filter(Boolean).filter(l=>!ignored(l)).sort().join('\n');
const gitStatusAfter=cp.execFileSync('git',['status','--short'],{cwd:repo,encoding:'utf8'});
if(filter(gitStatusAfter)!==filter(fs.readFileSync(path.join(dest,'git-status-before.txt'),'utf8')))throw Error('Changed outside scope');
const beforeCleanup=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(Object.keys(state.expected).some(p=>beforeCleanup[p]!==state.expected[p]))throw Error('Production mismatch before cleanup');
const junction=path.join(root,'work/node_modules');
if(fs.existsSync(junction))throw Error('Junction was not detached');
const depHashes=Object.fromEntries(Object.keys(summary.versions).map(p=>[p,sha(path.join(repo,'node_modules',p,'package.json'))]));
fs.copyFileSync(__filename,path.join(dest,'finish-cleanup.cjs.txt'));
fs.writeFileSync(path.join(dest,'cleanup-attempts.json'),JSON.stringify({first:{script:'finalize.cjs.txt',cwd:state.work,error:'EPERM at fs.rmSync(root); all pre-cleanup assertions passed; node_modules junction detached'},retry:{script:'finish-cleanup.cjs.txt',cwd:process.cwd(),target:root,targetVerifiedImmediateTempChild:true}},null,2));
if(fs.readdirSync(dest).some(n=>!n.endsWith('.txt')&&!n.endsWith('.json')))throw Error('Forbidden evidence suffix');
fs.rmSync(root,{recursive:true,force:true});
if(fs.existsSync(root))throw Error('Temp cleanup failed');
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(Object.keys(state.expected).some(p=>productionAfter[p]!==state.expected[p]))throw Error('Production mismatch after cleanup');
const sourceChangedAfterCleanup=Object.keys(before).filter(p=>sha(path.join(repo,p))!==before[p]);
if(sourceChangedAfterCleanup.length||Object.keys(depHashes).some(p=>sha(path.join(repo,'node_modules',p,'package.json'))!==depHashes[p]))throw Error('Source/dependencies changed during cleanup');
const links=[...section.matchAll(/\]\((spck-12-13-14-round7\/[^)]+)\)/g)].map(m=>m[1]);
const final={verifiedAt:new Date().toISOString(),head:state.head,verdict:summary.verdict,productionBefore:state.productionBefore,productionBeforeCleanup:beforeCleanup,productionAfter,productionMatchesOwner:true,sourceCount:state.sourceCount,srcScriptsCount:state.srcScriptsCount,snapshotComparedCount:state.sourceCount,snapshotComparedBeforeCleanup:true,sourcePathInventoryUnchanged:true,sourceChanged,snapshotChanged:summary.snapshotChanged,sourceChangedAfterCleanup,trackedDiffUnchanged:true,gitPathsOutsideAllowedScopeUnchanged:true,reportSha256:sha(reportPath),round1ThroughRound6HistoryByteUnchanged:true,originalHistoricalReportBytes:old.length,originalHistoricalReportSha256:sha(path.join(dest,'round6-report.md.txt')),originalProbesByteUnchanged:true,priorEvidenceFilesVerified:Object.keys(prior).length,priorEvidenceChanged,tempPath:root,tempRemoved:true,dependencyJunctionDetachedBeforeCleanup:true,dependencyDirectoryStillExists:fs.existsSync(path.join(repo,'node_modules')),dependencyPackageHashesPreserved:depHashes,cleanupFirstAttempt:'EPERM because cwd was inside snapshot; detached junction before attempt',cleanupRetry:'Succeeded from repository cwd after rechecking preservation',scope:'Only requested audit report and Round 7 evidence written; no product fixes, commit, push or real-service actions.'};
fs.writeFileSync(path.join(dest,'git-status-after.txt'),gitStatusAfter);
fs.writeFileSync(path.join(dest,'final-preservation.json'),JSON.stringify(final,null,2));
const hashes=Object.fromEntries(fs.readdirSync(dest).filter(n=>n!=='evidence-sha256.json'&&fs.statSync(path.join(dest,n)).isFile()).sort().map(n=>[n,sha(path.join(dest,n))]));
fs.writeFileSync(path.join(dest,'evidence-sha256.json'),JSON.stringify(hashes,null,2));
const missing=links.filter(p=>!fs.existsSync(path.join(repo,'docs/audits/v340',p)));
if(missing.length)throw Error('Missing evidence links: '+missing.join(','));
console.log(JSON.stringify(final,null,2));
console.log('Evidence files: '+fs.readdirSync(dest).length+'; all links exist.');
fs.unlinkSync(__filename);
