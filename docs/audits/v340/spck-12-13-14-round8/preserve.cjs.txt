const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const root=__dirname,state=JSON.parse(fs.readFileSync(path.join(root,'state.json'))),repo=state.repo,work=state.work;
const dest=path.join(repo,'docs/audits/v340/spck-12-13-14-round8');
if(fs.existsSync(dest))throw Error('Evidence destination exists');
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const before=JSON.parse(fs.readFileSync(path.join(root,'source-manifest-before.json')));
const after=Object.fromEntries(Object.keys(before).map(p=>[p,sha(path.join(repo,p))]));
const sourceChanged=Object.keys(before).filter(p=>before[p]!==after[p]);
const snapshotChanged=Object.keys(before).filter(p=>before[p]!==sha(path.join(work,p)));
const patch=cp.execFileSync('git',['diff','--binary','HEAD'],{cwd:repo,maxBuffer:20*1024*1024,stdio:['ignore','pipe','pipe']});
if(!patch.equals(fs.readFileSync(path.join(root,'working-tree.patch'))))throw Error('Original tracked diff changed');
const productionAfter=Object.fromEntries(Object.keys(state.expected).map(p=>[p,sha(path.join(repo,p))]));
if(sourceChanged.length||snapshotChanged.length||Object.keys(state.expected).some(p=>state.expected[p]!==productionAfter[p]))throw Error('Preservation mismatch');
fs.writeFileSync(path.join(root,'source-manifest-after.json'),JSON.stringify(after,null,2));
const gateNames=['tsc','v270','v272','s5','probe','supplementary','callback','focused','closure','review','round6','round7','context'];
const read=p=>JSON.parse(fs.readFileSync(path.join(root,p)));
const gates=gateNames.map(g=>read(g+'-result.json'));
const tests=Object.fromEntries([['round1','probe-results.json'],['round2','supplementary-results.json'],['F09','callback-results.json'],['round3','focused-results.json'],['round4','round4-closure-results.json'],['round5','round5-focused-results.json'],['round6','round6-focused-results.json'],['round7','round7-focused-results.json'],['round8','round8-context-results.json']].map(([k,p])=>{const r=read(p);return [k,{total:r.total,passed:r.passed,failed:r.failed,errors:r.errors??0}]}));
const statuses=Object.fromEntries(Array.from({length:10},(_,i)=>'F'+String(i+1).padStart(2,'0')).map(id=>[id,['F08'].includes(id)?'OPEN':'CLOSED']));
const mandatoryPassed=gates.slice(0,12).every(g=>g.exitCode===0);
const results=read('round8-context-results.json').results;
const verdict=mandatoryPassed&&results.every(r=>r.status==='PASS')?'GO':'NO-GO';
fs.writeFileSync(path.join(root,'closure-results.json'),JSON.stringify({verdict,closed:9,open:1,statuses,mandatoryPassed,mandatoryGateCount:11,mandatoryCommandCount:12,blockingProbes:results.filter(r=>r.status==='FAIL').map(r=>({id:r.id,finding:r.finding,label:r.label})),note:'All 95 original Round 1-7 probe cases pass; R7 classic-script repair confirmed. F01 closed. Three supplementary failures reproduce loss of .mjs module context in summary readers (folder, attachment, ZIP); F08 remains open. No product source edited.'},null,2));
for(const [round,filename] of [['round5','round5-focused-results.json'],['round6','round6-focused-results.json'],['round7','round7-focused-results.json'],['round8','round8-context-results.json']]){
 const observations=read(filename).results.map(r=>({id:r.id,finding:r.finding,status:r.status,label:r.label,...r.observed}));
 fs.writeFileSync(path.join(root,round+'-focused-observations.json'),JSON.stringify(observations,null,2));
}
const networkFiles=fs.readdirSync(root).filter(n=>n.endsWith('-network.jsonl'));
const events=networkFiles.flatMap(n=>fs.readFileSync(path.join(root,n),'utf8').trim().split('\n').filter(Boolean).map(line=>({log:n,...JSON.parse(line)})));
const versions=Object.fromEntries(['typescript','tsx','next','better-sqlite3','esbuild','playwright'].map(p=>[p,JSON.parse(fs.readFileSync(path.join(repo,'node_modules',p,'package.json'))).version]));
const reportPath=path.join(repo,'docs/audits/v340/CODEX_SPCK_12_13_14_AUDIT.md'),originalHistory=fs.readFileSync(path.join(root,'round7-report.md.txt'));
if(!fs.readFileSync(reportPath).equals(originalHistory))throw Error('Report changed unexpectedly');
const probeHashesAfter=Object.fromEntries(Object.entries(state.probeHashes).map(([name,pair])=>{const p=name.endsWith('.ts')?path.join(work,'scratch',name):path.join(root,name);const actual=sha(p);if(actual!==pair.source||pair.source!==pair.executed)throw Error('Original probe changed '+name);return [name,actual]}));
const walk=(dir,rel='')=>fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(dir,e.name),rel+e.name+'/'):[rel+e.name]);
const priorEvidence=read('prior-evidence-manifest.json'); if(Object.keys(priorEvidence).some(p=>sha(path.join(repo,p))!==priorEvidence[p]))throw Error('Prior evidence changed');
fs.writeFileSync(path.join(root,'prior-evidence-manifest.json'),JSON.stringify(priorEvidence,null,2));
const summary={...state,verdict,mandatoryGateCount:11,mandatoryCommandCount:12,mandatoryPassed,gates,tests,statuses,versions,productionAfter,productionMatchesOwner:true,sourceChanged,snapshotChanged,trackedDiffUnchanged:true,probeHashesAfter,round6ProbeSha256:sha(path.join(work,'scratch/round6-focused-probe.ts')),history:{originalBytes:originalHistory.length,originalSha256:sha(path.join(root,'round7-report.md.txt'))},priorEvidenceFiles:Object.keys(priorEvidence).length,network:{logs:networkFiles,events:events.length,blocked:events.filter(e=>e.kind.startsWith('blocked')),sqliteOpened:events.filter(e=>e.kind==='sqlite-open').map(e=>e.host),limitation:'In-process instrumentation only; not an operating-system packet capture'},reviewer:{agent:'/root',scope:'Independent auditor: source review plus actual synthetic runtime probes in TEMP; no separate reviewer or unavailable model execution claimed'},scope:'Only requested audit report and Round 8 evidence written. Synthetic/mocked services; no source edits, commit, push or real-service acceptance.'};
fs.writeFileSync(path.join(root,'verification-summary.json'),JSON.stringify(summary,null,2));
for(const name of ['independent-probe.ts','supplementary-probe.ts','round3-focused-probe.ts','round4-closure-probe.ts','round5-focused-probe.ts','round6-focused-probe.ts','round7-focused-probe.ts','round8-context-probe.ts'])fs.copyFileSync(path.join(work,'scratch',name),path.join(root,name+'.txt'));
fs.writeFileSync(path.join(root,'review-source-anchors.txt'),cp.execFileSync('rg',['-n','analyzeJsCodeSyntax|sourceType|acorn|readZipRelJsPathsByZip|sameZipEntries|activeForbiddenPatterns|inspectZipArchiveSafely|htmlInlineStyles|blockMatch|prefixWithExt|mergedLinksMap|normStuName','src/app/api/spck/scan/route.ts','src/lib/gemini/spck-comment.ts','src/lib/playwright/spck-uploader.ts'],{cwd:work,encoding:'utf8'}));

fs.mkdirSync(dest);
const outputName=n=>/\.(?:txt|json)$/.test(n)?n:n+'.txt';
for(const name of fs.readdirSync(root).filter(n=>fs.statSync(path.join(root,n)).isFile())){fs.copyFileSync(path.join(root,name),path.join(dest,outputName(name)));if(sha(path.join(root,name))!==sha(path.join(dest,outputName(name))))throw Error('Evidence copy mismatch '+name);}
console.log(JSON.stringify({dest,verdict,mandatoryPassed,tests,statuses,sourceCount:state.sourceCount,sourceChanged,snapshotChanged,blockedEvents:summary.network.blocked,versions,history:summary.history},null,2));
